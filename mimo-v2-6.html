Hello World Guide

Run MiMo-V2.6-Pro-RL locally using Hugging Face transformers or vLLM.

Python
from transformers import AutoModelForCausalLM, AutoTokenizer
import torch

model_id = "XiaomiMiMo/MiMo-V2.6-Pro-RL"

tokenizer = AutoTokenizer.from_pretrained(model_id, trust_remote_code=True)
model = AutoModelForCausalLM.from_pretrained(
    model_id,
    torch_dtype=torch.bfloat16,
    device_map="auto",
    trust_remote_code=True
)

prompt = "Analyze this complex algorithmic puzzle, verify constraints, and synthesize a proof."
messages = [
    {"role": "system", "content": "You are MiMo-V2.6-Pro, an omnimodal RL-trained reasoning assistant."},
    {"role": "user", "content": prompt}
]

text = tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
inputs = tokenizer([text], return_tensors="pt").to(model.device)

outputs = model.generate(
    **inputs,
    max_new_tokens=1024,
    temperature=0.6,
    top_p=0.95
)

response = tokenizer.decode(outputs[0][inputs.input_ids.shape[1]:], skip_special_tokens=True)
print(response)
